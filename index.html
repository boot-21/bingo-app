from telegram import Update, ReplyKeyboardMarkup, KeyboardButton, WebAppInfo from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, ContextTypes, filters # የቦት ቶከንዎን ከ BotFather የተሰጠውን እዚህ ያስገቡ BOT_TOKEN = "YOUR_TELEGRAM_BOT_TOKEN" # የ GitHub Pages ሊንክዎን እዚህ ያስገቡ (የ index.html አድራሻ) WEB_APP_URL = "https://dittbrukernavn.github.io/ditt-repositorium/" async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): # Register ቁልፍ ላይ request_contact=True በመጠቀም ስልክ ቁጥር እንዲልኩ ይደረጋል keyboard = [ [ KeyboardButton("Play 🎮", web_app=WebAppInfo(url=WEB_APP_URL)), KeyboardButton("Register 📝", request_contact=True) ], [ KeyboardButton("Check Balance 💵"), KeyboardButton("Deposit 💵") ], [ KeyboardButton("Contact Support ☎️"), KeyboardButton("Instruction 📖") ], [ KeyboardButton("Transfer 🎁"), KeyboardButton("Withdraw 🤑") ], [ KeyboardButton("Invite 🔗"), KeyboardButton("Convert Bonus 💸") ] ] reply_markup = ReplyKeyboardMarkup(keyboard, resize_keyboard=True) await update.message.reply_text( "👋 Welcome to Beteseb Bingo! Choose an Option below.", reply_markup=reply_markup ) # ተጫዋቹ ስልክ ቁጥሩን ሲያጋራ (Share Contact) የሚቀበለው ክፍል async def contact_handler(update: Update, context: ContextTypes.DEFAULT_TYPE): contact = update.message.contact if contact: phone_number = contact.phone_number await update.message.reply_text( f"✅ አመሰግናለሁ! ስልክ ቁጥርዎ ({phone_number}) በተሳካ ሁኔታ ተመዝግቧል።" ) # የጽሁፍ መልዕክቶችን እና የ Deposit ሂደትን የሚያስተዳድረው ክፍል async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE): text = update.message.text user_data = context.user_data if text == "Deposit 💵": user_data['waiting_deposit'] = True await update.message.reply_text( "💰 ማስገባት የሚፈልጉትን መጠን ከ10 ብር ጀምሮ ያስገቡ (ለምሳሌ፦ 50):" ) elif user_data.get('waiting_deposit'): user_data['waiting_deposit'] = False amount = text await update.message.reply_text( f"📥 የጠየቁት መጠን: {amount} ብር\n\n" f"እባክዎን ከዚህ በታች ባለው የቴሌብር አካውንት ገንዘቡን ያስተላልፉ:\n" f"📱 **Telebirr Account:** `0903123027`\n\n" f"ገንዘቡን ካስተላለፉ በኋላ የክፍያ ማረጋገጫ (Receipt) ለአድሚን ይላኩ።" ) elif text == "Check Balance 💵": await update.message.reply_text("💵 የእርስዎ ቀሪ ሂሳብ: 0.00 ብር") elif text == "Contact Support ☎️": await update.message.reply_text("☎️ ለድጋፍ ሰጪ @Admin_Username ያነጋገሩ።") elif text == "Instruction 📖": await update.message.reply_text("📖 የቢንጎ ጨዋታ መመሪያ እዚህ ይገኛል...") if __name__ == '__main__': app = ApplicationBuilder().token(BOT_TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(MessageHandler(filters.CONTACT, contact_handler)) app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message)) print("ቦቱ በስራ ላይ ነው...") app.run_polling()
